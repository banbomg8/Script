local function fastPress(key)
   pcall(function()
      VirtualInputManager:SendKeyEvent(true, key, false, game)
      task.wait(0.015)
      VirtualInputManager:SendKeyEvent(false, key, false, game)
   end)
end

end

local comboCooldown = 0
local waterAngle = 0

local function isInSafeZone(playerChar)
   if not playerChar then return true end
   if playerChar:FindFirstChild("SafeZone") then return true end
   local hrp = playerChar:FindFirstChild("HumanoidRootPart")
   if hrp then
      for _, obj in pairs(Workspace:GetChildren()) do
         if obj.Name:lower():find("safe") or obj.Name:lower():find("zone") then
            if obj:IsA("Part") or obj:IsA("Model") then
               local pos = obj:IsA("Part") and obj.Position or (obj.PrimaryPart and obj.PrimaryPart.Position)
               if pos and (hrp.Position - pos).Magnitude < 150 then
                  return true
               end
            end
         end
      end
   end
   return false
end

RunService.RenderStepped:Connect(function()
   local char = LocalPlayer.Character
   if not char then return end
   local hrp = char:FindFirstChild("HumanoidRootPart")
   local humanoid = char:FindFirstChild("Humanoid")
   
   -- تفعيل الـ Noclip المنفصل
   if noclipActive then
      for _, part in pairs(char:GetDescendants()) do
         if part:IsA("BasePart") then
            part.CanCollide = false
         end
      end
   end
   
   if waterBoxActive and realWaterPart and hrp and humanoid then
      waterAngle = waterAngle + 0.05
      local offset = CFrame.new(math.cos(waterAngle) * 4, 1.5, math.sin(waterAngle) * 4)
      realWaterPart.CFrame = hrp.CFrame * offset
      
      humanoid.PlatformStand = true
      pcall(function()
         humanoid:ChangeState(Enum.HumanoidStateType.Swimming)
      end)
   end
   
   if trueInfHpActive and humanoid then
      pcall(function()
         humanoid.MaxHealth = math.huge
         humanoid.Health = math.huge
      end)
   end
   
   if antiStunActive and humanoid and not waterBoxActive and not bountyHuntActive then
      pcall(function()
         humanoid.Sit = false
         humanoid.PlatformStand = false
      end)
   end
   
   if tacticalHpActive and humanoid and hrp then
      if humanoid.Health <= 3000 and not isRetreating then
         isRetreating = true
      elseif isRetreating and humanoid.Health >= 6000 then
         isRetreating = false
      end
   else
      isRetreating = false
   end
   
   if hardAimActive and hrp then
      local closestTarget = nil
      local shortestDist = math.huge
      for _, player in pairs(Players:GetPlayers()) do
         if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local targetHrp = player.Character.HumanoidRootPart
            local dist = (hrp.Position - targetHrp.Position).Magnitude
            if dist < shortestDist then
               shortestDist = dist
               closestTarget = targetHrp
            end
         end
      end
      if closestTarget then
         Workspace.CurrentCamera.CFrame = CFrame.new(Workspace.CurrentCamera.CFrame.Position, closestTarget.Position + Vector3.new(0, 2, 0))
      end
   end
   
   if bountyHuntActive and hrp then
      pcall(function()
         if not char:FindFirstChild("HasBuso") then
            triggerBusoHaki()
         end
      end)
      
      if isRetreating then
         humanoid.PlatformStand = true
         pcall(function() humanoid:ChangeState(Enum.HumanoidStateType.Swimming) end)
         hrp.Velocity = Vector3.new(0, 0, 0)
         hrp.CFrame = hrp.CFrame + Vector3.new(0, 4.5, 0)
         return
      end
      
      if not currentTarget or not currentTarget.Parent or not currentTarget.Parent:FindFirstChild("Humanoid") or currentTarget.Parent.Humanoid.Health <= 0 or isInSafeZone(currentTarget.Parent) then
         local closestTarget = nil
         local shortestDist = math.huge
         for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
               local pChar = p.Character
               if not isInSafeZone(pChar) then
                  local dist = (hrp.Position - pChar.HumanoidRootPart.Position).Magnitude
                  if dist < shortestDist then
                     shortestDist = dist
                     closestTarget = pChar.HumanoidRootPart
                  end
               end
            end
         end
         currentTarget = closestTarget
      end
      
      if currentTarget then
         humanoid.PlatformStand = true
         pcall(function() humanoid:ChangeState(Enum.HumanoidStateType.Swimming) end)
         
         local distance = (currentTarget.Position - hrp.Position).Magnitude
         
         if distance > 5 then
            hrp.Velocity = Vector3.new(0, 0, 0)
            local targetPos = currentTarget.Position + Vector3.new(0, 2.5, 0)
            local direction = (targetPos - hrp.Position).Unit
            local flightSpeed = 190
            local step = math.min(flightSpeed * RunService.RenderStepped:Wait(), distance)
            hrp.CFrame = hrp.CFrame + (direction * step)
         else
            hrp.Velocity = Vector3.new(0, 0, 0)
            
            if tick() - comboCooldown > 2.5 then
               comboCooldown = tick()
               
               task.spawn(function()
                  fastPress(Enum.KeyCode.Two)
                  task.wait(0.015)
                  
                  VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.V, false, game)
                  task.wait(3.0)
                  VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.V, false, game)
                  task.wait(0.015)
                  
                  fastPress(Enum.KeyCode.C)
                  task.wait(0.015)
                  
                  fastPress(Enum.KeyCode.X)
                  task.wait(0.015)
                  
                  fastPress(Enum.KeyCode.One)
                  task.wait(0.015)
                  
                  fastPress(Enum.KeyCode.Z)
                  task.wait(0.015)
                  
                  fastPress(Enum.KeyCode.C)
                  task.wait(0.015)
                  
                  fastPress(Enum.KeyCode.X)
               end)
            end
         end
      end
   elseif not bountyHuntActive and not waterBoxActive and humanoid and humanoid.PlatformStand then
      humanoid.PlatformStand = false
   end
end)
